package content

import (
	"encoding/json"
	"io"

	"github.com/samber/oops"
)

type IconMapType map[string]any

func (i IconMapType) MarshalJSON() ([]byte, error) {
	b, err := json.Marshal(map[string]any(i))
	if err != nil {
		return nil, err
	}
	return json.Marshal(string(b))
}

func (i *IconMapType) UnmarshalJSON(b []byte) error {
	var s string
	if err := json.Unmarshal(b, &s); err != nil {
		// If it isn't a JSON string literal, assume it's already unquoted.
		var m map[string]any
		if err2 := json.Unmarshal(b, &m); err2 != nil {
			return oops.Wrapf(err2, "IconMapType: cannot unmarshal data")
		}
		*i = IconMapType(m)
		return nil
	}

	var m map[string]any
	if err := json.Unmarshal([]byte(s), &m); err != nil {
		return oops.Wrapf(err, "IconMapType: cannot unmarshal inner string content")
	}

	*i = IconMapType(m)
	return nil
}

func (i *IconMapType) UnmarshalGQL(v interface{}) error {
	m, ok := v.(map[string]interface{})
	if !ok {
		return oops.New("GraphQL type assertion failed")
	}
	*i = IconMapType(m)
	return nil
}

func (i IconMapType) MarshalGQL(w io.Writer) {
	b, err := json.Marshal(map[string]any(i))
	if err != nil {
		panic("cannot marshal IconMapType: " + err.Error())
	}
	_, err = w.Write(b)
	if err != nil {
		panic("cannot write marshalled IconMapType: " + err.Error())
	}
}

package lessonreport

import (
	"context"
	"errors"
	"strings"
)

type LessonReport struct {
	TodaysLessons          []string `json:"todaysLessons"`
	ProgressAndEngagements []string `json:"progressAndEngagements"`
	UpcomingLearnings      []string `json:"upcomingLearnings"`
}

type Repository interface {
	GenerateLessonReport(ctx context.Context, input GenerateReportInput) (*LessonReport, error)
}

type GenerateReportInput struct {
	StudentName           string `json:"studentName"`
	TodaysLesson          string `json:"todaysLesson"`
	ProgressAndEngagement string `json:"progressAndEngagement"`
	UpcomingLearnings     string `json:"upcomingLearnings"`
}

func (i *GenerateReportInput) FormatAndValidate() error {
	i.StudentName = strings.TrimSpace(i.StudentName)
	i.TodaysLesson = strings.TrimSpace(i.TodaysLesson)
	i.ProgressAndEngagement = strings.TrimSpace(i.ProgressAndEngagement)
	i.UpcomingLearnings = strings.TrimSpace(i.UpcomingLearnings)
	return i.validate()
}

var (
	ErrEmptyStudentName        = errors.New("student name is empty")
	ErrEmptyTodaysLessonPrompt = errors.New("today's lesson prompt is empty")
)

func (i GenerateReportInput) validate() error {
	if len(i.StudentName) == 0 {
		return ErrEmptyStudentName
	}
	if len(i.TodaysLesson) == 0 {
		return ErrEmptyTodaysLessonPrompt
	}
	return nil
}

// GetResetConfirmationInput compares the fields of two questions and generates a QuestionConfirmationInput for resetting the confirmation state.
func GetResetConfirmationInput(oldQuestion, currentQuestion Question, changedType bool) QuestionConfirmationInput {
	input := QuestionConfirmationInput{}
	boolPtr := func(b bool) *bool { return &b }
	// safely compare two *string values, accounting for nil pointers
	unequalStringP := func(a, b *string) *bool {
		if a == nil || b == nil {
			return boolPtr(a != b)
		}
		return boolPtr(*a != *b)
	}
	// On question type change, answer is reset
	if changedType {
		input.AnswerCorrect = boolPtr(true)
	}
	input.Grammar = boolPtr(oldQuestion.Base().Content != currentQuestion.Base().Content || oldQuestion.Base().Hint != currentQuestion.Base().Hint)
	input.Figure = unequalStringP(oldQuestion.Base().Figure, currentQuestion.Base().Figure)

	if oldShortQuestion, ok := oldQuestion.(*ShortQuestion); ok {
		currentShortQuestion := currentQuestion.(*ShortQuestion)
		input.Prefix = unequalStringP(oldShortQuestion.Prefix, currentShortQuestion.Prefix)
		input.Suffix = unequalStringP(oldShortQuestion.Suffix, currentShortQuestion.Suffix)
	}
	return input
}

func CollectConfirmsEdges(confirmations []*edges.UserConfirmsQuestionEdge) QuestionConfirmation {
	confirmation := QuestionConfirmation{}
	for _, edge := range confirmations {
		confirmation.Grammar += edge.Grammar
		confirmation.AnswerCorrect += edge.AnswerCorrect
		confirmation.Prefix += edge.Prefix
		confirmation.Suffix += edge.Suffix
		confirmation.Figure += edge.Figure
	}
	return confirmation
}

func InputToConfirmationEdge(userID, questionID id.ID, input QuestionConfirmationInput) edges.UserConfirmsQuestionEdge {
	toInt := func(b *bool) int {
		if b == nil {
			return 0
		} else if *b {
			return 1
		} else {
			return -1
		}
	}

	return edges.UserConfirmsQuestionEdge{
		Edge:          edge.Edge{FromID: userID, ToID: questionID},
		Grammar:       toInt(input.Grammar),
		AnswerCorrect: toInt(input.AnswerCorrect),
		Prefix:        toInt(input.Prefix),
		Suffix:        toInt(input.Suffix),
		Figure:        toInt(input.Figure),
	}
}

func GetActiveQuestionCacheKey(questionID id.ID) string {
	return fmt.Sprintf("activeQuestion_%s", questionID)
}

func CollectConfirmsEdges(confirmations []*edges.UserConfirmsSkillEdge) SkillConfirmation {
	confirmation := SkillConfirmation{}
	for _, edge := range confirmations {
		confirmation.Title += edge.Title
		confirmation.Description += edge.Description
		confirmation.TitleGrammar += edge.TitleGrammar
		confirmation.DescriptionGrammar += edge.DescriptionGrammar
	}
	return confirmation
}

func InputToConfirmationEdge(userID, skillID id.ID, input SkillConfirmationInput) edges.UserConfirmsSkillEdge {
	toInt := func(b *bool) int {
		if b == nil {
			return 0
		} else if *b {
			return 1
		} else {
			return -1
		}
	}
	return edges.UserConfirmsSkillEdge{
		Edge:               edge.Edge{FromID: userID, ToID: skillID},
		Title:              toInt(input.Title),
		Description:        toInt(input.Description),
		TitleGrammar:       toInt(input.TitleGrammar),
		DescriptionGrammar: toInt(input.DescriptionGrammar),
	}
}

func (f SearchSkillsFilters) AsQdrantFilter() *qdrant.Filter {
	conditions := []*qdrant.Condition{}

	if f.SubtopicID != nil {
		conditions = append(conditions, &qdrant.Condition{
			ConditionOneOf: &qdrant.Condition_Field{
				Field: &qdrant.FieldCondition{
					Key: "subtopics",
					Match: &qdrant.Match{
						MatchValue: &qdrant.Match_Keyword{
							Keyword: *f.SubtopicID,
						},
					},
				},
			},
		})
	}

	if f.TopicID != nil {
		conditions = append(conditions, &qdrant.Condition{
			ConditionOneOf: &qdrant.Condition_Field{
				Field: &qdrant.FieldCondition{
					Key: "topics",
					Match: &qdrant.Match{
						MatchValue: &qdrant.Match_Keyword{
							Keyword: *f.TopicID,
						},
					},
				},
			},
		})
	}

	if f.CurriculumID != nil {
		conditions = append(conditions, &qdrant.Condition{
			ConditionOneOf: &qdrant.Condition_Field{
				Field: &qdrant.FieldCondition{
					Key: "curriculums",
					Match: &qdrant.Match{
						MatchValue: &qdrant.Match_Keyword{
							Keyword: *f.CurriculumID,
						},
					},
				},
			},
		})
	}

	if f.NetworkID != nil {
		conditions = append(conditions, &qdrant.Condition{
			ConditionOneOf: &qdrant.Condition_Field{
				Field: &qdrant.FieldCondition{
					Key: "networks",
					Match: &qdrant.Match{
						MatchValue: &qdrant.Match_Keyword{
							Keyword: *f.NetworkID,
						},
					},
				},
			},
		})
	}

	if f.HasNotesAndExamples != nil {
		conditions = append(conditions, &qdrant.Condition{
			ConditionOneOf: &qdrant.Condition_Field{
				Field: &qdrant.FieldCondition{
					Key: "hasNotesAndExamples",
					Match: &qdrant.Match{
						MatchValue: &qdrant.Match_Boolean{
							Boolean: *f.HasNotesAndExamples,
						},
					},
				},
			},
		})
	}

	if f.IsPTo10 != nil {
		conditions = append(conditions, &qdrant.Condition{
			ConditionOneOf: &qdrant.Condition_Field{
				Field: &qdrant.FieldCondition{
					Key: "isPTo10",
					Match: &qdrant.Match{
						MatchValue: &qdrant.Match_Boolean{
							Boolean: *f.IsPTo10,
						},
					},
				},
			},
		})
	}

	return &qdrant.Filter{Must: conditions}
}

)

var (
	_ BaseTopic = (*Strand)(nil)
	_ BaseTopic = (*Topic)(nil)
)

type (
	BaseTopic interface {
		IsEntity()
		IsBaseTopic()
		Base() *BaseTopicNode
		GetID() id.ID
	}
	BaseTopicNode struct {
		node.Abstract `baseType:"BaseTopic"`
		Title         string `json:"title"`
		Color         string `json:"color"`
		Emoji         string `json:"emoji"`
		UUID          string `json:"uuid" neo4j:"optional"`
	}
	Strand struct {
		node.Node `neo4j:"Strand,id:strand"`
		BaseTopicNode
	}
	Topic struct {
		node.Node `neo4j:"Topic,id:topic"`
		BaseTopicNode
		Description string `json:"description"`
		Icon        string `json:"icon"`
		Difficulty  int    `json:"difficulty"`
	}
)

func (BaseTopicNode) Implementers() []neogo.IAbstract {
	return []neogo.IAbstract{
		&Strand{},
		&Topic{},
	}
}
func (BaseTopicNode) IsBaseTopic() {}

func (s Strand) GetID() id.ID { return s.ID }
func (s Strand) Base() *BaseTopicNode {
	return &s.BaseTopicNode
}

func (t Topic) GetID() id.ID { return t.ID }
func (t Topic) Base() *BaseTopicNode {
	return &t.BaseTopicNode
}
