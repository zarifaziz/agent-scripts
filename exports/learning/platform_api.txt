type Progress struct {
	Correct float64 `json:"correct"`
	Wrong   float64 `json:"wrong"`
}

func (p *Progress) Add(progress Progress) {
	p.Correct += progress.Correct
	p.Wrong += progress.Wrong
}

func (p *Progress) Divide(divisor float64) {
	p.Correct /= divisor
	p.Wrong /= divisor
}

package platformsubtopic

import (
	"context"

	learningv1 "buf.build/gen/go/tutero/learning/protocolbuffers/go/v1/learning"
	"github.com/samber/oops"

	platformedges "github.com/MathGaps/learning/v2/api/platform/edges"
)

var enumToSubtopicStateMap = map[learningv1.SubtopicState]platformedges.SubtopicState{
	learningv1.SubtopicState_SUBTOPIC_STATE_COMPLETE:           platformedges.SubtopicStateComplete,
	learningv1.SubtopicState_SUBTOPIC_STATE_TEACHING:           platformedges.SubtopicStateTeaching,
	learningv1.SubtopicState_SUBTOPIC_STATE_RETURN_LATER:       platformedges.SubtopicStateReturnLater,
	learningv1.SubtopicState_SUBTOPIC_STATE_READY_TO_TEACH:     platformedges.SubtopicStateReadyToTeach,
	learningv1.SubtopicState_SUBTOPIC_STATE_LESSON_NOT_PLANNED: platformedges.SubtopicStateLessonNotPlanned,
	learningv1.SubtopicState_SUBTOPIC_STATE_PROCESSING:         platformedges.SubtopicStateProcessing,
}

func GetEnumFromSubtopicState(ctx context.Context, state platformedges.SubtopicState) (learningv1.SubtopicState, error) {
	for key, val := range enumToSubtopicStateMap {
		if state == val {
			return key, nil
		}
	}
	return -1, oops.
		With("state", string(state)).
		New("invalid subtopic state")
}

func GetSubtopicStateFromEnum(ctx context.Context, enum learningv1.SubtopicState) (platformedges.SubtopicState, error) {
	subtopicState, ok := enumToSubtopicStateMap[enum]
	if !ok {
		return "", oops.
			With("enum", enum.String()).
			New("invalid enum")
	}
	return subtopicState, nil
}
