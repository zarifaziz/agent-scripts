package resource

import (

	"github.com/MathGaps/core/pkg/domain/id"
	"github.com/MathGaps/resources/api/common"

type (
	DiagnosticTestSessionStatus         string
	DiagnosticTestSessionAssessmentType string
)

const (
	DTSessionStatusCreated  DiagnosticTestSessionStatus = "CREATED"
	DTSessionStatusStarted  DiagnosticTestSessionStatus = "STARTED"
	DTSessionStatusPaused   DiagnosticTestSessionStatus = "PAUSED"
	DTSessionStatusFinished DiagnosticTestSessionStatus = "FINISHED"

	DTSessionDiagnosticAssessment DiagnosticTestSessionAssessmentType = "DIAGNOSTIC"
	DTSessionReviewAssessment     DiagnosticTestSessionAssessmentType = "REVIEW"
)

type DiagnosticTestSession struct {
	BaseResource `neo4j:"DiagnosticTestSession,id:dts"`
	CreatedAt    time.Time `json:"createdAt"`
	CreatedBy    id.ID     `json:"createdBy"`

	DueDate             time.Time                           `json:"dueDate"`
	StartedAt           time.Time                           `json:"startedAt"`
	FinishedAt          time.Time                           `json:"finishedAt"`
	Status              DiagnosticTestSessionStatus         `json:"status"`
	InitialProgress     *common.SessionProgress             `json:"initialProgress"`
	FinalProgress       *common.SessionProgress             `json:"finalProgress"`
	LastPausedAt        time.Time                           `json:"lastPausedAt"`
	TotalPausedDuration time.Duration                       `json:"totalPausedDuration"`
	TimeTaken           time.Duration                       `json:"timeTaken"`
	TimeSaved           time.Duration                       `json:"timeSaved"`
	AssessmentType      DiagnosticTestSessionAssessmentType `json:"assessmentType"`

	TopologyGenerated bool `json:"topologyGenerated" neo4j:"optional"`
}

func (dts *DiagnosticTestSession) ComputeTimeTaken() time.Duration {
	return time.Duration(dts.FinishedAt.Sub(dts.StartedAt) - dts.TotalPausedDuration)
}

func (dts *DiagnosticTestSession) TimeElapsed() time.Duration {
	if dts.Status == DTSessionStatusFinished {
		return dts.TimeTaken
	}
	switch dts.Status {
	case DTSessionStatusCreated:
		return 0
	case DTSessionStatusStarted:
		return time.Duration(time.Since(dts.StartedAt) - dts.TotalPausedDuration)
	case DTSessionStatusPaused:
		return time.Duration(dts.LastPausedAt.Sub(dts.StartedAt) - dts.TotalPausedDuration)
	case DTSessionStatusFinished:
		return dts.TimeTaken
	default:
		return 0
	}
}

func (dts *DiagnosticTestSession) ComputeTimeSaved(numSubtopics int) time.Duration {
	// We don't have access to skills (without a gRPC call), so we make a rough estimate based off # subtopics.
	randWithin := func(min, max float64) float64 {
		return min + rand.Float64()*(max-min)
	}
	randMins := float64(numSubtopics) * randWithin(17, 25)
	return time.Duration(math.Max(randMins, 20) * float64(time.Minute))
}

type DiagnosticTestSessionPreview struct {
	DiagnosticTestSession

	ClassID     id.ID   `json:"classID"`
	TopicIDs    []id.ID `json:"topicIDs"`
	SubtopicIDs []id.ID `json:"subtopicIDs"`
	Path        string  `json:"path"`
}

func (dts *DiagnosticTestSessionPreview) GetPath() string { return dts.Path }

func (f *SearchResourcesFilter) ToQdrantFilter() *qdrant.Filter {
	must := []*qdrant.Condition{}

	if len(f.ClassIDs) > 0 {
		must = append(must, &qdrant.Condition{
			ConditionOneOf: &qdrant.Condition_Field{
				Field: &qdrant.FieldCondition{
					Key: "classID",
					Match: &qdrant.Match{
						MatchValue: &qdrant.Match_Keywords{
							Keywords: &qdrant.RepeatedStrings{
								Strings: f.ClassIDs,
							},
						},
					},
				},
			},
		})
	}

	if f.ResourceType != "" {
		must = append(must, &qdrant.Condition{
			ConditionOneOf: &qdrant.Condition_Field{
				Field: &qdrant.FieldCondition{
					Key: "resourceType",
					Match: &qdrant.Match{
						MatchValue: &qdrant.Match_Text{
							Text: string(f.ResourceType),
						},
					},
				},
			},
		})
	}

	if f.TopicID != "" {
		must = append(must, &qdrant.Condition{
			ConditionOneOf: &qdrant.Condition_Field{
				Field: &qdrant.FieldCondition{
					Key: "topicIDs",
					Match: &qdrant.Match{
						MatchValue: &qdrant.Match_Keyword{
							Keyword: string(f.TopicID),
						},
					},
				},
			},
		})
	}

	if f.Path != "" {
		path := common.EnsurePathFormat(f.Path)
		if f.Recursive {
			must = append(must, &qdrant.Condition{
				ConditionOneOf: &qdrant.Condition_Field{
					Field: &qdrant.FieldCondition{
						Key: "path",
						Match: &qdrant.Match{
							MatchValue: &qdrant.Match_Text{
								Text: path,
							},
						},
					},
				},
			})
		} else {
			must = append(must, qdrant.NewMatch("path", path))
		}
	}

	return &qdrant.Filter{Must: must}
}

func (i UpsertResourcePreviewInput) QdrantPayload() (map[string]*qdrant.Value, error) {
	if i.UUID == "" {
		return nil, oops.New("UUID is required to upsert resource preview")
	}
	if i.ResourceType == "" {
		return nil, oops.New("ResourceType is required to upsert resource preview")
	}

	payload := map[string]*qdrant.Value{}
	payload["resourceType"] = qdrant.NewValueString(string(i.ResourceType))

	switch i.ResourceType {
	case ResourceTypeLessonPlan, ResourceTypeWorksheet:
		if i.WithLessonPlan == nil {
			return nil, oops.New("WithLessonPlan input is required for lesson plan preview")
		}
		return i.buildLessonPlanPayload(i.ResourceType, payload)
	case ResourceTypeDiagnosticTestSession:
		if i.WithDiagnosticTest == nil {
			return nil, oops.New("WithDiagnosticTest input is required for diagnostic test session preview")
		}
		return i.buildDiagnosticTestPayload(payload)
	case ResourceTypeDocument:
		if i.WithDocument == nil {
			return nil, oops.New("WithDocument input is required for document preview")
		}
		return i.buildDocumentPayload(payload)
	case ResourceTypeUpload:
		if i.WithUpload == nil {
			return nil, oops.New("WithUpload input is required for upload preview")
		}
		return i.buildUploadPayload(payload)
	default:
		return nil, oops.New("unsupported resource type for preview upsert")
	}
}

func (i UpsertResourcePreviewInput) buildLessonPlanPayload(resourceType ResourceType, basePayload map[string]*qdrant.Value) (map[string]*qdrant.Value, error) {
	lp := i.WithLessonPlan

	if lp.ResourceID != "" {
		basePayload["id"] = qdrant.NewValueString(lp.ResourceID)
	}
	if lp.Name != nil && *lp.Name != "" {
		basePayload["name"] = qdrant.NewValueString(*lp.Name)
	}
	if lp.Path != nil {
		basePayload["path"] = qdrant.NewValueString(*lp.Path)
	}
	if lp.TopicID != nil && *lp.TopicID != "" {
		basePayload["topicID"] = qdrant.NewValueString(*lp.TopicID)
	}
	if lp.SubtopicID != nil && *lp.SubtopicID != "" {
		basePayload["subtopicID"] = qdrant.NewValueString(*lp.SubtopicID)
	}
	if lp.LessonsCount != nil {
		basePayload["lessonsCount"] = qdrant.NewValueInt(int64(*lp.LessonsCount))
	}
	if lp.ClassID != nil {
		basePayload["classID"] = qdrant.NewValueString(*lp.ClassID)
	}
	if lp.CreatedAt != nil {
		basePayload["createdAt"] = qdrant.NewValueString(lp.CreatedAt.Format(time.RFC3339))
	}
	if lp.LastViewedAt != nil {
		basePayload["lastViewedAt"] = qdrant.NewValueString(lp.LastViewedAt.Format(time.RFC3339))
	}
	if len(lp.TopicIDs) > 0 {
		basePayload["topicIDs"] = qdrant.NewValueList(&qdrant.ListValue{
			Values: lo.Map(lp.TopicIDs, func(id id.ID, _ int) *qdrant.Value {
				return qdrant.NewValueString(id)
			}),
		})
	}
	if len(lp.SubtopicIDs) > 0 {
		basePayload["subtopicIDs"] = qdrant.NewValueList(&qdrant.ListValue{
			Values: lo.Map(lp.SubtopicIDs, func(id id.ID, _ int) *qdrant.Value {
				return qdrant.NewValueString(id)
			}),
		})
	}
	if lp.PortraitCoverImage != nil && *lp.PortraitCoverImage != "" {
		basePayload["portraitCoverImage"] = qdrant.NewValueString(*lp.PortraitCoverImage)
	}
	if lp.IntroductionImageURL != nil && *lp.IntroductionImageURL != "" {
		basePayload["introductionImageURL"] = qdrant.NewValueString(*lp.IntroductionImageURL)
	}
	if lp.CoverImageGenerated != nil {
		basePayload["coverImageGenerated"] = qdrant.NewValueBool(*lp.CoverImageGenerated)
	}

	// Lesson plan specific fields
	if resourceType == ResourceTypeLessonPlan {
		if lp.Context != nil {
			basePayload["context"] = qdrant.NewValueString(*lp.Context)
		}
		if lp.ContextIcon != nil {
			basePayload["contextIcon"] = qdrant.NewValueString(*lp.ContextIcon)
		}
		if lp.LessonType != nil {
			basePayload["lessonType"] = qdrant.NewValueString(string(*lp.LessonType))
		}
	}

	if resourceType == ResourceTypeWorksheet {
		if lp.WorksheetPlanType != nil {
			basePayload["worksheetPlanType"] = qdrant.NewValueString(string(*lp.WorksheetPlanType))
		}
	}

	return basePayload, nil
}

func (i UpsertResourcePreviewInput) buildDiagnosticTestPayload(basePayload map[string]*qdrant.Value) (map[string]*qdrant.Value, error) {
	dt := i.WithDiagnosticTest

	if dt.ID != "" {
		basePayload["id"] = qdrant.NewValueString(dt.ID)
	}
	if dt.Name != nil && *dt.Name != "" {
		basePayload["name"] = qdrant.NewValueString(*dt.Name)
	}
	if dt.Path != "" {
		basePayload["path"] = qdrant.NewValueString(dt.Path)
	}
	if dt.Status != "" {
		basePayload["status"] = qdrant.NewValueString(string(dt.Status))
	}
	if dt.DueDate != nil {
		basePayload["dueDate"] = qdrant.NewValueString(dt.DueDate.Format(time.RFC3339))
	}
	if dt.StartedAt != nil {
		basePayload["startedAt"] = qdrant.NewValueString(dt.StartedAt.Format(time.RFC3339))
	}
	if dt.CreatedAt != nil {
		basePayload["createdAt"] = qdrant.NewValueString(dt.CreatedAt.Format(time.RFC3339))
	}
	if dt.LastViewedAt != nil {
		basePayload["lastViewedAt"] = qdrant.NewValueString(dt.LastViewedAt.Format(time.RFC3339))
	}
	if dt.TimeTaken != nil {
		basePayload["timeTaken"] = qdrant.NewValueInt(int64(*dt.TimeTaken))
	}
	if dt.AssessmentType != "" {
		basePayload["assessmentType"] = qdrant.NewValueString(string(dt.AssessmentType))
	}
	if dt.ClassID != "" {
		basePayload["classID"] = qdrant.NewValueString(dt.ClassID)
	}
	if len(dt.TopicIDs) > 0 {
		basePayload["topicIDs"] = qdrant.NewValueList(&qdrant.ListValue{
			Values: lo.Map(dt.TopicIDs, func(id id.ID, _ int) *qdrant.Value {
				return qdrant.NewValueString(id)
			}),
		})
	}
	if len(dt.SubtopicIDs) > 0 {
		basePayload["subtopicIDs"] = qdrant.NewValueList(&qdrant.ListValue{
			Values: lo.Map(dt.SubtopicIDs, func(id id.ID, _ int) *qdrant.Value {
				return qdrant.NewValueString(id)
			}),
		})
	}

	return basePayload, nil
}

type Resource interface {
	neogo.IAbstract

	IsEntity()
	IsResource()
	GetName() string
	GetResourceType() ResourceType
	Base() *BaseResource
}

type ResourcePreview interface {
	neogo.IAbstract

	IsEntity()
	IsResource()
	GetName() string
	GetResourceType() ResourceType
	GetPath() string
	Base() *BaseResource
}

type BaseResource struct {
	neogo.Abstract `neo4j:"Resource"`
	neogo.Node
	Name         string       `json:"name"`
	Type         ResourceType `json:"resourceType"`
	UUID         string       `json:"uuid"`
	LastViewedAt time.Time    `json:"lastViewedAt"`
}

func (r BaseResource) IsEntity()                     {}
func (r BaseResource) IsResource()                   {}
func (r BaseResource) GetName() string               { return r.Name }
func (r BaseResource) GetResourceType() ResourceType { return r.Type }
func (r *BaseResource) Base() *BaseResource          { return r }

func (BaseResource) Implementers() []neogo.IAbstract {
	return []neogo.IAbstract{
		&LessonPlan{},
		&WorksheetPlan{},
		&DiagnosticTestSession{},
		&Document{},
	}
}

package resource_test

import (
	"testing"
	"time"

	"github.com/MathGaps/core/pkg/domain/id"
	"github.com/MathGaps/resources/api/resource"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

func TestBuildUploadPayload(t *testing.T) {
	t.Parallel()

	t.Run("builds complete upload payload", func(t *testing.T) {
		t.Parallel()

		now := time.Now()
		resourceID := id.MustNew()
		classID := id.MustNew()
		uploadType := "VIDEO"
		name := "Test Video"
		url := "https://youtube.com/watch?v=test"
		thumbnail := "https://img.youtube.com/vi/test/default.jpg"
		path := "~/folder1/"

		input := resource.UpsertResourcePreviewInput{
			UUID:              "test-uuid-1234",
			ResourceType:      resource.ResourceTypeUpload,
			ShouldCreatePoint: true,
			WithUpload: &resource.WithUploadInput{
				ResourceID:   resourceID,
				Name:         &name,
				ClassID:      &classID,
				CreatedAt:    &now,
				Path:         &path,
				LastViewedAt: &now,
				UploadType:   &uploadType,
				URL:          &url,
				Thumbnail:    &thumbnail,
			},
		}

		payload, err := input.QdrantPayload()
		require.NoError(t, err)

		assert.Equal(t, string(resource.ResourceTypeUpload), payload["resourceType"].GetStringValue())
		assert.Equal(t, string(resourceID), payload["id"].GetStringValue())
		assert.Equal(t, name, payload["name"].GetStringValue())
		assert.Equal(t, string(classID), payload["classID"].GetStringValue())
		assert.Equal(t, path, payload["path"].GetStringValue())
		assert.Equal(t, uploadType, payload["uploadType"].GetStringValue())
		assert.Equal(t, url, payload["url"].GetStringValue())
		assert.Equal(t, thumbnail, payload["thumbnail"].GetStringValue())
		assert.NotEmpty(t, payload["createdAt"].GetStringValue())
		assert.NotEmpty(t, payload["lastViewedAt"].GetStringValue())
	})

	t.Run("builds upload payload without optional thumbnail", func(t *testing.T) {
		t.Parallel()

		now := time.Now()
		resourceID := id.MustNew()
		classID := id.MustNew()
		uploadType := "LINK"
		name := "Test Link"
		url := "https://example.com"
		path := "~/"

		input := resource.UpsertResourcePreviewInput{
			UUID:              "test-uuid-5678",
			ResourceType:      resource.ResourceTypeUpload,
			ShouldCreatePoint: true,
			WithUpload: &resource.WithUploadInput{
				ResourceID:   resourceID,
				Name:         &name,
				ClassID:      &classID,
				CreatedAt:    &now,
				Path:         &path,
				LastViewedAt: &now,
				UploadType:   &uploadType,
				URL:          &url,
				Thumbnail:    nil,
			},
		}

		payload, err := input.QdrantPayload()
		require.NoError(t, err)

		assert.Equal(t, uploadType, payload["uploadType"].GetStringValue())
		assert.Equal(t, url, payload["url"].GetStringValue())
		_, hasThumbnail := payload["thumbnail"]
		assert.False(t, hasThumbnail, "thumbnail should not be present when nil")
	})

	t.Run("returns error when WithUpload is nil", func(t *testing.T) {
		t.Parallel()

		input := resource.UpsertResourcePreviewInput{
			UUID:              "test-uuid-9999",
			ResourceType:      resource.ResourceTypeUpload,
			ShouldCreatePoint: true,
			WithUpload:        nil,
		}

		_, err := input.QdrantPayload()
		require.Error(t, err)
		assert.Contains(t, err.Error(), "WithUpload input is required for upload preview")
	})

	t.Run("returns error when UUID is empty", func(t *testing.T) {
		t.Parallel()

		input := resource.UpsertResourcePreviewInput{
			UUID:              "",
			ResourceType:      resource.ResourceTypeUpload,
			ShouldCreatePoint: true,
			WithUpload:        &resource.WithUploadInput{},
		}

		_, err := input.QdrantPayload()
		require.Error(t, err)
		assert.Contains(t, err.Error(), "UUID is required")
	})

	t.Run("returns error when ResourceType is empty", func(t *testing.T) {
		t.Parallel()

		input := resource.UpsertResourcePreviewInput{
			UUID:              "test-uuid",
			ResourceType:      "",
			ShouldCreatePoint: true,
			WithUpload:        &resource.WithUploadInput{},
		}

		_, err := input.QdrantPayload()
		require.Error(t, err)
		assert.Contains(t, err.Error(), "ResourceType is required")
	})
}
