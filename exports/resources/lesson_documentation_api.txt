package lessondocumentation

import (
	"reflect"

	"github.com/MathGaps/core/pkg/domain/id"
)

var lpnTypeMap = map[reflect.Type]DocumentationNodeType{
	reflect.TypeOf(&ContentDescriptorsNode{}):         DocumentationNodeTypeContentDescriptors,
	reflect.TypeOf(&SuccessCriteriaNode{}):            DocumentationNodeTypeSuccessCriteria,
	reflect.TypeOf(&LearningObjectivesNode{}):         DocumentationNodeTypeLearningObjectives,
	reflect.TypeOf(&WarmUpQuestionsNode{}):            DocumentationNodeTypeWarmUpQuestions,
	reflect.TypeOf(&WarmUpQuestionsWithContextNode{}): DocumentationNodeTypeWarmUpQuestionsWithContextNode,
	reflect.TypeOf(&ThoughtSparkerNode{}):             DocumentationNodeTypeThoughtSparker,
	reflect.TypeOf(&ApplicationNode{}):                DocumentationNodeTypeApplication,
	reflect.TypeOf(&BrainTeaserNode{}):                DocumentationNodeTypeBrainTeaser,
	reflect.TypeOf(&MathsDebateNode{}):                DocumentationNodeTypeMathsDebate,
	reflect.TypeOf(&ExitTicketNode{}):                 DocumentationNodeTypeExitTicket,
	reflect.TypeOf(&ContemplativeQuestionNode{}):      DocumentationNodeTypeContemplativeQuestion,
	reflect.TypeOf(&NoteNode{}):                       DocumentationNodeTypeNote,
	reflect.TypeOf(&ActivityNode{}):                   DocumentationNodeTypeActivity,
	reflect.TypeOf(&SituationalQuestionNode{}):        DocumentationNodeTypeSituationalQuestion,
}

func ldn[N any, PN interface {
	LessonNode
	*N
}](modifiers ...func(PN),
) PN {
	node := PN(new(N))
	base := node.Base()
	base.SetID(id.MustNew())
	base.Enabled = false
	base.Hidden = false
	base.Ready = false
	nodeType := reflect.TypeOf(node)
	base.Type = lpnTypeMap[nodeType]
	base.Name = node.GetDefaultName()
	for _, m := range modifiers {
		m(node)
	}
	return node
}

func GetLessonDocumentationNodes() (_ []LessonNode) {
	return []LessonNode{
		ldn[ContentDescriptorsNode](),
		ldn[SuccessCriteriaNode](),
		ldn[LearningObjectivesNode](),
		ldn[WarmUpQuestionsNode](),
		ldn[ThoughtSparkerNode](),
		ldn[ApplicationNode](),
		ldn[BrainTeaserNode](),
		ldn[MathsDebateNode](),
		ldn[ActivityNode](),
		ldn[ExitTicketNode](),
		ldn[ContemplativeQuestionNode](),
		ldn[SituationalQuestionNode](),
		ldn[WarmUpQuestionsWithContextNode](),
	}
}

// GetLessonDocumentationNoteNode creates a new node type of Note
// Used whenever user creates a custom documentation node.
func GetLessonDocumentationNoteNode() (_ *NoteNode) {
	node := ldn[NoteNode]()
	base := node.Base()
	base.Enabled = true
	base.Hidden = false
	base.Ready = true
	return node
}
