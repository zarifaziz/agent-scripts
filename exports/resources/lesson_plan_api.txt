var LessonPlanSkillDuration time.Duration = min(5)

type LessonPlanConfig interface {
	IsEnabled() bool
	IsExcluded() bool
	SetDuration(time.Duration)
}

type LessonPlanConfigs []LessonPlanConfig

type LessonPlanNodeConfig struct {
	neogo.Node `neo4j:"LessonPlanNodeConfig"`
	CreatedAt  time.Time `json:"createdAt"`

	LessonPlanNode       `json:"-"`
	GroupDuration        *time.Duration `json:"groupDuration,omitempty"`
	FlexibleFromDuration *time.Duration `json:"flexibleFromDuration,omitempty"`
	CanToggle            bool           `json:"canToggle"`
	Excluded             bool           `json:"-"`
}

type LessonPlanNodeGroup struct {
	neogo.Node
	CreatedAt time.Time `json:"createdAt"`

	Enabled  bool                    `json:"enabled"`
	Type     LessonPlanNodeGroupType `json:"type"`
	Duration time.Duration           `json:"duration"`
}

type LessonPlanGroupConfig struct {
	LessonPlanNodeGroup `neo4j:"LessonPlanGroupConfig"`
	Nodes               []LessonPlanNodeConfig `json:"-"`
	CanToggle           bool                   `json:"canToggle"`
}

func (l *LessonPlanNodeConfig) IsEnabled() bool {
	return l.Base().Enabled
}

func (l *LessonPlanNodeConfig) IsExcluded() bool {
	return l.Excluded
}

func (l *LessonPlanNodeConfig) SetDuration(duration time.Duration) {
	if l.GroupDuration != nil {
		*l.GroupDuration = duration
	}
	l.Base().Duration = duration
}

func (l *LessonPlanGroupConfig) IsEnabled() bool {
	return l.Enabled
}

func (l *LessonPlanGroupConfig) IsExcluded() bool {
	return false
}

func (l *LessonPlanGroupConfig) SetDuration(duration time.Duration) {
	l.Duration = duration
	for _, node := range l.Nodes {
		node.SetDuration(duration)
	}
}

type SectionConfig struct {
	Duration          time.Duration     `json:"duration"`
	LessonPlanConfigs LessonPlanConfigs `json:"lessonPlanConfigs"`
}
type SectionConfigs struct {
	IntroductionConfigs SectionConfig `json:"introductionConfigs"`
	PracticeConfigs     SectionConfig `json:"practiceConfigs"`
}

func (sc *SectionConfigs) Configs() LessonPlanConfigs {
	return append(sc.IntroductionConfigs.LessonPlanConfigs, sc.PracticeConfigs.LessonPlanConfigs...)
}

func (sc *SectionConfigs) GroupConfigs() []LessonPlanGroupConfig {
	groupConfigs := sc.IntroductionConfigs.LessonPlanConfigs.GroupConfigs()
	groupConfigs = append(groupConfigs, sc.PracticeConfigs.LessonPlanConfigs.GroupConfigs()...)
	return groupConfigs
}

func (sc *SectionConfigs) SetDuration(duration time.Duration) {
	sc.IntroductionConfigs.Duration = duration
	sc.PracticeConfigs.Duration = duration
	for _, config := range sc.Configs() {
		config.SetDuration(duration)
	}
	for _, groupConfig := range sc.GroupConfigs() {
		groupConfig.SetDuration(duration)
	}
}

// We have to distribute the lesson duration among the sections based on the nodes included in the sections
// If lesson doesn't have any nodes, we distribute the total duration equally among the sections
// If certain sections have nodes, we distribute the remaining duration among the sections with nodes
func (sc *SectionConfigs) FlexDurations(lessonDuration time.Duration) {
	totalConcreteDuration := sc.IntroductionConfigs.Duration + sc.PracticeConfigs.Duration

	sectionsToDistributeDuration := map[*SectionConfig]bool{}
	if sc.IntroductionConfigs.Duration > 0 {
		sectionsToDistributeDuration[&sc.IntroductionConfigs] = true
	}
	if sc.PracticeConfigs.Duration > 0 {
		sectionsToDistributeDuration[&sc.PracticeConfigs] = true
	}

	if len(sectionsToDistributeDuration) == 0 {
		sectionsToDistributeDuration = map[*SectionConfig]bool{
			&sc.IntroductionConfigs: true,
			&sc.PracticeConfigs:     true,
		}
	}

	// the section duration should not be in decimals i.e 10.56 minutes should be converted to 11 minutes
	numOfSectionsToDistributeDuration := len(sectionsToDistributeDuration)
	remainingDuration := lessonDuration - totalConcreteDuration
	for sectionConfig := range sectionsToDistributeDuration {
		durationToDistribute := remainingDuration / time.Duration(numOfSectionsToDistributeDuration)
		durationWithoutDecimals := int(math.Ceil(durationToDistribute.Minutes()))
		durationToDistributeInMinutes := time.Duration(durationWithoutDecimals) * time.Minute
		durationToDistribute = time.Duration(durationToDistributeInMinutes.Nanoseconds())
		sectionConfig.Duration += durationToDistribute
		remainingDuration -= durationToDistribute
		numOfSectionsToDistributeDuration -= 1
	}
}

func sectionConfigs(nc nodeConfigsToInclude, lessonDuration time.Duration) *SectionConfigs {
	nodeIndex := -1
	introConfigs := LessonPlanConfigs{}
	practiceConfigs := LessonPlanConfigs{}

	introConfigs = append(introConfigs, config(lpn[IntroductionNode](nc.IntroductionNode.Enabled, min(1), increment(&nodeIndex)), nc.IntroductionNode.Excluded))
	introConfigs = append(introConfigs, config(lpn[LessonAgendaNode](nc.LessonAgenda.Enabled, min(2), increment(&nodeIndex)), nc.LessonAgenda.Excluded))
	introConfigs = append(introConfigs, config(lpn[LearningGoalsNode](nc.LearningGoals.Enabled, min(2), increment(&nodeIndex)), nc.LearningGoals.Excluded))
	introConfigs = append(introConfigs,
		group(nc.LessonStarter.Enabled, LessonPlanNodeGroupLessonStarter, []*LessonPlanNodeConfig{
			config(lpn[WarmUpQuestionsNode](nc.WarmUpQuestionsNode.Enabled, sec(150), increment(&nodeIndex)), nc.WarmUpQuestionsNode.Excluded),
			config(lpn[WarmUpQuestionsWithContextNode](nc.WarmUpQuestionsWithContextNode.Enabled, sec(150), increment(&nodeIndex)), nc.WarmUpQuestionsWithContextNode.Excluded),
		}, true),
	)
	introConfigs = append(introConfigs, config(lpn[ApplicationNode](nc.ApplicationNode.Enabled, min(3), increment(&nodeIndex)), nc.ApplicationNode.Excluded))

	activityDuration := min(7)
	if lessonDuration < min(35) {
		activityDuration = min(5)
	}
	practiceConfigs = append(practiceConfigs, config(lpn[ActivityNode](nc.Activity.Enabled, activityDuration, increment(&nodeIndex)), nc.Activity.Excluded))

	situationalQuestionDuration := min(7)
	if lessonDuration < min(35) {
		situationalQuestionDuration = min(5)
	}
	practiceConfigs = append(practiceConfigs, config(lpn[SituationalQuestionNode](nc.SituationalQuestionNode.Enabled, situationalQuestionDuration, increment(&nodeIndex)), nc.SituationalQuestionNode.Excluded))

	misconceptionQuestionDuration := min(7)
	if lessonDuration < min(35) {
		misconceptionQuestionDuration = min(5)
	}
	practiceConfigs = append(practiceConfigs, config(lpn[MisconceptionQuestionNode](nc.MisconceptionQuestionNode.Enabled, misconceptionQuestionDuration, increment(&nodeIndex)), nc.MisconceptionQuestionNode.Excluded))

	practiceConfigs = append(practiceConfigs,
		group(nc.WrapUpExercise.Enabled, LessonPlanNodeGroupWrapUpExercise, []*LessonPlanNodeConfig{
			func() *LessonPlanNodeConfig {
				contemplative := lpn[ContemplativeQuestionNode](nc.ContemplativeQuestionNode.Enabled, sec(120), increment(&nodeIndex))
				contemplative.Difficulty = DifficultyMild
				return config(contemplative, nc.ContemplativeQuestionNode.Excluded)
			}(),
			config(lpn[ThoughtSparkerNode](nc.ThoughtSparkerNode.Enabled, sec(150), increment(&nodeIndex)), nc.ThoughtSparkerNode.Excluded),
			config(lpn[ExitTicketNode](nc.ExitTicketNode.Enabled, sec(120), increment(&nodeIndex)), nc.ExitTicketNode.Excluded),
		}, true))

	// Filter out excluded configs
	enabledIntroConfigs := LessonPlanConfigs{}
	for _, cfg := range introConfigs {
		if !cfg.IsExcluded() {
			enabledIntroConfigs = append(enabledIntroConfigs, cfg)
		}
	}

	enabledPracticeConfigs := LessonPlanConfigs{}
	for _, cfg := range practiceConfigs {
		if !cfg.IsExcluded() {
			enabledPracticeConfigs = append(enabledPracticeConfigs, cfg)
		}
	}

	return &SectionConfigs{
		IntroductionConfigs: SectionConfig{
			LessonPlanConfigs: enabledIntroConfigs,
		},
		PracticeConfigs: SectionConfig{
			LessonPlanConfigs: enabledPracticeConfigs,
		},
	}
}

type NodeStatus struct {
	Excluded bool
	Enabled  bool
}
type nodeConfigsToInclude struct {
	ThoughtSparkerNode             NodeStatus
	WarmUpQuestionsNode            NodeStatus
	WarmUpQuestionsWithContextNode NodeStatus
	ApplicationNode                NodeStatus
	LessonStarter                  NodeStatus
	Skills                         NodeStatus
	LessonAgenda                   NodeStatus
	LearningGoals                  NodeStatus
	ContemplativeQuestionNode      NodeStatus
	ExitTicketNode                 NodeStatus
	Activity                       NodeStatus
	WrapUpExercise                 NodeStatus
	SituationalQuestionNode        NodeStatus
	MisconceptionQuestionNode      NodeStatus
	IntroductionNode               NodeStatus
}

var (
	includedAndEnabled  = NodeStatus{Enabled: true}
	includedButDisabled = NodeStatus{Enabled: false}
	excludedAndDisabled = NodeStatus{Excluded: true}
)

func (l *CreateLessonPlanInput) GetLessonConfigs() ([]*SectionConfigs, error) {
	configs := make([]*SectionConfigs, len(l.Durations))

	nodeConfig := nodeConfigsToInclude{}
	for _, node := range l.IncludeNodes {
		switch node {
		case LessonPlanNodeTypeThoughtSparker:
			nodeConfig.ThoughtSparkerNode = includedAndEnabled
		case LessonPlanNodeTypeWarmUpQuestions:
			nodeConfig.WarmUpQuestionsNode = includedAndEnabled
		case LessonPlanNodeTypeWarmUpQuestionsWithContext:
			nodeConfig.WarmUpQuestionsWithContextNode = includedAndEnabled
		case LessonPlanNodeTypeApplication:
			nodeConfig.ApplicationNode = includedAndEnabled
		case LessonPlanNodeTypeLessonAgenda:
			nodeConfig.LessonAgenda = includedAndEnabled
		case LessonPlanNodeTypeLearningGoals:
			nodeConfig.LearningGoals = includedAndEnabled
		case LessonPlanNodeTypeContemplativeQuestion:
			nodeConfig.ContemplativeQuestionNode = includedAndEnabled
		case LessonPlanNodeTypeActivity:
			nodeConfig.Activity = includedAndEnabled
		case LessonPlanNodeTypeExitTicket:
			nodeConfig.ExitTicketNode = includedAndEnabled
		case LessonPlanNodeTypeSituationalQuestion:
			nodeConfig.SituationalQuestionNode = includedAndEnabled
		case LessonPlanNodeTypeMisconceptionQuestion:
			nodeConfig.MisconceptionQuestionNode = includedAndEnabled
		case LessonPlanNodeTypeIntroduction:
			nodeConfig.IntroductionNode = includedAndEnabled
		}
	}

	// always keep the lesson starter and wrap up exercise nodes in disabled state
	// and toggle it later if included
	nodeConfig.LessonStarter = excludedAndDisabled
	nodeConfig.WrapUpExercise = excludedAndDisabled

	for lessonIndex, duration := range l.Durations {
		nc := nodeConfig

		// we assign the skills uniformly to the lessons. So if the lesson index is less than the number of skills,
		// we have enough skills to assign to the lesson
		enoughSkills := len(l.SkillIDs) > lessonIndex
		nc.Skills = NodeStatus{Enabled: enoughSkills}
		if !nc.Skills.Enabled {
			nc.LessonAgenda = includedButDisabled
			nc.LearningGoals = includedButDisabled
		}

		// include application node and introduction node only in the first lesson
		if lessonIndex != 0 {
			nc.ApplicationNode = excludedAndDisabled
			nc.ThoughtSparkerNode = excludedAndDisabled
			nc.IntroductionNode = excludedAndDisabled
		}

		// include the following only in the last lesson
		if lessonIndex != len(l.Durations)-1 {
			nc.Activity = excludedAndDisabled
			nc.SituationalQuestionNode = excludedAndDisabled
		}

		// include node only in even number periods (2nd, 4th, etc.)
		// unless there's only one period
		if len(l.Durations) > 1 && lessonIndex%2 != 0 {
			nc.MisconceptionQuestionNode = excludedAndDisabled
		}
		sectionConfigs := sectionConfigs(nc, duration)
		if duration == 0 {
			sectionConfigs.SetDuration(0)
		} else {
			introductionDuration := sectionConfigs.IntroductionConfigs.LessonPlanConfigs.TotalDuration()
			practiceDuration := sectionConfigs.PracticeConfigs.LessonPlanConfigs.TotalDuration()
			sectionConfigs.IntroductionConfigs.Duration = introductionDuration
			sectionConfigs.PracticeConfigs.Duration = practiceDuration
			sectionConfigs.FlexDurations(duration)
		}
		configs[lessonIndex] = sectionConfigs
	}
	return configs, nil
}

func sec(n int) time.Duration {
	return time.Duration(n) * time.Second
}

func min(n int) time.Duration {
	return time.Duration(n) * time.Minute
}

func config(lpn LessonPlanNode, excluded bool) *LessonPlanNodeConfig {
	node := neogo.Node{ID: id.MustNew()}
	return &LessonPlanNodeConfig{
		Node:           node,
		LessonPlanNode: lpn,
		CanToggle:      true,
		Excluded:       excluded,
	}
}

func group(enabled bool, groupType LessonPlanNodeGroupType, lpns []*LessonPlanNodeConfig, canToggle bool) *LessonPlanGroupConfig {
	node := neogo.Node{ID: id.MustNew()}
	var (
		groupDuration time.Duration
		nodes         []LessonPlanNodeConfig
	)
	for _, lpn := range lpns {
		duration := lpn.Base().Duration
		groupDuration += duration
		lpn.GroupDuration = &duration
		nodes = append(nodes, *lpn)
	}
	return &LessonPlanGroupConfig{
		LessonPlanNodeGroup: LessonPlanNodeGroup{
			Enabled:  enabled,
			Node:     node,
			Type:     groupType,
			Duration: groupDuration,
		},
		Nodes:     nodes,
		CanToggle: canToggle,
	}
}

func lpn[N any, PN interface {
	LessonPlanNode
	*N
}](enabled bool, duration time.Duration, index int, modifiers ...func(PN),
) PN {
	node := PN(new(N))
	base := node.Base()
	base.SetID(id.MustNew())
	base.Enabled = enabled
	base.Duration = duration
	base.Index = index
	nodeType := reflect.TypeOf(node)
	base.Type = lpnTypeMap[nodeType]
	for _, m := range modifiers {
		m(node)
	}
	return node
}

func (n LessonPlanConfigs) FlexToDuration(desiredDuration time.Duration) (err error) {
	var totalDuration, flexibleDuration time.Duration
	flexibleNodes := map[int]time.Duration{}
	allNodeConfigs := map[int]*LessonPlanNodeConfig{}
	i := -1
	for _, config := range n {
		// GroupConfigs
		if gc, ok := config.(*LessonPlanGroupConfig); ok {
			for _, node := range gc.Nodes {
				i++
				allNodeConfigs[i] = &node
			}

			if gc.IsEnabled() {
				totalDuration += gc.Duration
				var enabledGD time.Duration
				for _, node := range gc.Nodes {
					if node.IsEnabled() {
						enabledGD += *node.GroupDuration
					}
				}
				if enabledGD == 0 {
					continue
				}
				for _, node := range gc.Nodes {
					if !node.IsEnabled() {
						continue
					}
					ratio := float64(*node.GroupDuration) / float64(enabledGD)
					duration := ratio * float64(gc.Duration)
					node.Base().Duration = time.Duration(duration)
				}
			}
		}

		// NodeConfigs
		if nc, ok := config.(*LessonPlanNodeConfig); ok {
			i++
			allNodeConfigs[i] = nc
			if !nc.IsEnabled() {
				continue
			}
			lpnDuration := nc.Base().Duration
			totalDuration += lpnDuration
			if nc.FlexibleFromDuration != nil {
				flexibleDuration += lpnDuration
				flexibleNodes[i] = lpnDuration
			}
		}
	}

	flex := func(
		flexibleNodes map[int]time.Duration,
		total, flexible, flexTo time.Duration,
	) error {
		remaining := flexTo - total
		if len(flexibleNodes) == 0 {
			return nil
		}
		lastIdx := 0
		for i, d := range flexibleNodes {
			var ratio time.Duration
			if flexible == 0 {
				ratio = 1
			} else {
				ratio = d / flexible
			}
			delta := (flexTo - total) * ratio
			allNodeConfigs[i].Base().Duration += delta
			if allNodeConfigs[i].Base().Duration < *allNodeConfigs[i].FlexibleFromDuration {
				return ErrReduceDurationBeyondConfigMinimum
			}
			remaining -= delta
			lastIdx = i
		}
		if remaining > 0 {
			allNodeConfigs[lastIdx].Base().Duration += remaining
		}
		return nil
	}
	return flex(flexibleNodes, totalDuration, flexibleDuration, desiredDuration)
}

func (n LessonPlanConfigs) TotalDuration() time.Duration {
	sum := time.Duration(0)
	for _, config := range n {
		if gc, ok := config.(*LessonPlanGroupConfig); ok {
			if !gc.Enabled {
				continue
			}
			sum += gc.Duration
		} else if nc, ok := config.(*LessonPlanNodeConfig); ok {
			if !nc.Base().Enabled {
				continue
			}
			sum += nc.Base().Duration
		}
	}
	return sum
}

func (n LessonPlanConfigs) EnabledNodes() []LessonPlanNode {
	nodes := []LessonPlanNode{}
	for _, config := range n {
		if gc, ok := config.(*LessonPlanGroupConfig); ok {
			for _, node := range gc.Nodes {
				if !node.Base().Enabled {
					continue
				}
				nodes = append(nodes, node.LessonPlanNode)
			}
		} else if nc, ok := config.(*LessonPlanNodeConfig); ok {
			if !nc.Base().Enabled {
				continue
			}
			nodes = append(nodes, nc.LessonPlanNode)
		}
	}
	return nodes
}

func (n LessonPlanConfigs) Nodes() []LessonPlanNode {
	nodes := []LessonPlanNode{}
	for _, config := range n {
		if gc, ok := config.(*LessonPlanGroupConfig); ok {
			for _, node := range gc.Nodes {
				nodes = append(nodes, node.LessonPlanNode)
			}
		} else if nc, ok := config.(*LessonPlanNodeConfig); ok {
			nodes = append(nodes, nc.LessonPlanNode)
		}
	}
	return nodes
}

func (n LessonPlanConfigs) NodeConfigs() []LessonPlanNodeConfig {
	configs := []LessonPlanNodeConfig{}
	for _, config := range n {
		if gc, ok := config.(*LessonPlanGroupConfig); ok {
			configs = append(configs, gc.Nodes...)
		} else if nc, ok := config.(*LessonPlanNodeConfig); ok {
			configs = append(configs, *nc)
		}
	}
	return configs
}

func (n LessonPlanConfigs) GroupConfigs() []LessonPlanGroupConfig {
	groupConfigs := []LessonPlanGroupConfig{}
	for _, config := range n {
		if gc, ok := config.(*LessonPlanGroupConfig); ok {
			groupConfigs = append(groupConfigs, *gc)
		}
	}
	return groupConfigs
}

type WorksheetPlanSectionType string

const (
	WorksheetPlanSectionTypeIntroduction WorksheetPlanSectionType = "IntroductionSection"
	WorksheetPlanSectionTypeEnabling     WorksheetPlanSectionType = "EnablingSection"
	WorksheetPlanSectionTypeExtension    WorksheetPlanSectionType = "ExtensionSection"
	WorksheetPlanSectionTypeSkill        WorksheetPlanSectionType = "SkillSection"
)

var OrderedWorksheetSectionTypes = []WorksheetPlanSectionType{
	WorksheetPlanSectionTypeIntroduction,
	WorksheetPlanSectionTypeEnabling,
	WorksheetPlanSectionTypeExtension,
	WorksheetPlanSectionTypeSkill,
}

type WorksheetPlanSectionConfigsInput struct {
	NodeType   LessonPlanNodeType
	Difficulty Difficulty
	Count      int
	Type       WorksheetPlanSectionType
}

func WorksheetPlanSectionConfigs(nodeConfigs []WorksheetPlanSectionConfigsInput) *WorksheetSectionConfigs {
	nodeIndex := -1
	defaultDuration := min(7)
	var (
		introductionConfigs LessonPlanConfigs
		enablingConfigs     LessonPlanConfigs
		extensionConfigs    LessonPlanConfigs
	)

	addQuestionToSection := func(configs *LessonPlanConfigs, count int, difficulty Difficulty) {
		for range count {
			node := lpn[ScaffoldedQuestionNode](true, defaultDuration, increment(&nodeIndex))
			node.Difficulty = difficulty
			*configs = append(*configs, config(node, true))
		}
	}

	for _, nc := range nodeConfigs {
		switch nc.NodeType {
		case LessonPlanNodeTypeScaffoldedQuestion:
			switch nc.Type {
			case WorksheetPlanSectionTypeEnabling:
				addQuestionToSection(&enablingConfigs, nc.Count, nc.Difficulty)
			case WorksheetPlanSectionTypeExtension:
				addQuestionToSection(&extensionConfigs, nc.Count, nc.Difficulty)
			}
		case LessonPlanNodeTypeLearningGoals:
			for range nc.Count {
				node := lpn[LearningGoalsNode](true, min(2), increment(&nodeIndex))
				introductionConfigs = append(introductionConfigs, config(node, true))
			}
		}
	}

	return &WorksheetSectionConfigs{
		IntroductionConfigs: SectionConfig{
			LessonPlanConfigs: introductionConfigs,
		},
		EnablingConfigs: SectionConfig{
			LessonPlanConfigs: enablingConfigs,
		},
		ExtensionConfigs: SectionConfig{
			LessonPlanConfigs: extensionConfigs,
		},
	}
}

func (l *CreateLessonPlanInput) GetWorksheetConfigs(regeneration bool) ([]*WorksheetSectionConfigs, error) {
	if !lo.Contains(resource.WroksheetPlanTypes, l.WorksheetPlanConfiguration.Type) {
		return nil, oops.Errorf("unsupported worksheet plan type: %s", l.WorksheetPlanConfiguration.Type)
	}
	switch l.WorksheetPlanConfiguration.Type {
	case resource.WorksheetPlanTypeEnabling, resource.WorksheetPlanTypeExtension:
		return l.GetEnablingExtensionWorksheetConfigs()
	default:
		return l.GetStandardWorksheetConfigs(regeneration)
	}
}

func (l *CreateLessonPlanInput) GetStandardWorksheetConfigs(regeneration bool) ([]*WorksheetSectionConfigs, error) {
	nodeConfig := worksheetNodeConfigsToInclude{}
	for nodeType, count := range l.WorksheetPlanConfiguration.NodeCounts {
		if nodeType == LessonPlanNodeTypeScaffoldedQuestion {
			nodeConfig.ScaffoldedQuestionNode = count
		}
	}

	configs := make([]*WorksheetSectionConfigs, len(l.Durations))

	wpc := l.WorksheetPlanConfiguration
	minDiff, maxDiff := WorksheetEditDifficultyEnabling, WorksheetEditDifficultyExt
	if regeneration {
		minDiff, maxDiff = WorksheetEditDifficultyEasy, WorksheetEditDifficultyMedium
	}
	includeEnabling := wpc.MinDifficulty == nil || *wpc.MinDifficulty <= int(minDiff)
	includeExtension := wpc.MaxDifficulty == nil || *wpc.MaxDifficulty >= int(maxDiff)

	createSectionConfig := CreateWorksheetConfigCreator(nodeConfig.ScaffoldedQuestionNode, ConfigCreatorOptions{
		IncludeEnabling:  includeEnabling,
		IncludeExtension: includeExtension,
		MinDifficulty:    wpc.MinDifficulty,
		MaxDifficulty:    wpc.MaxDifficulty,
	})

	for lessonIndex := range l.Durations {
		nodeConfigs := []WorksheetPlanSectionConfigsInput{
			{NodeType: LessonPlanNodeTypeLearningGoals, Count: 1},
			createSectionConfig(WorksheetPlanSectionTypeEnabling),
			createSectionConfig(WorksheetPlanSectionTypeExtension),
		}
		configs[lessonIndex] = WorksheetPlanSectionConfigs(nodeConfigs)
	}
	return configs, nil
}

func (l *CreateLessonPlanInput) GetEnablingExtensionWorksheetConfigs() ([]*WorksheetSectionConfigs, error) {
	nodeConfigs := [][]LessonPlanNodeConfig{}
	defaultDuration := min(7)
	nodeIndex := -1
	introductionConfigs := LessonPlanConfigs{
		config(lpn[LearningGoalsNode](true, min(2), increment(&nodeIndex)), true),
	}
	skillSectionCount := len(l.SkillIDs)
	scaffoldedCount := l.WorksheetPlanConfiguration.NodeCounts[LessonPlanNodeTypeScaffoldedQuestion]
	baseCount := scaffoldedCount / skillSectionCount
	remainderCount := scaffoldedCount % skillSectionCount
	difficulty := DifficultyEnabling
	if l.WorksheetPlanConfiguration.Type == resource.WorksheetPlanTypeExtension {
		difficulty = DifficultyExtension
	}
	for i := 0; i < skillSectionCount; i++ {
		nodeConfigs = append(nodeConfigs, []LessonPlanNodeConfig{})
		// give each section the "base" count,
		scaffoldedQuestionCount := baseCount
		// and one extra for the last rem sections
		if i >= skillSectionCount-remainderCount {
			scaffoldedQuestionCount++
		}
		for j := 0; j < scaffoldedQuestionCount; j++ {
			node := lpn[ScaffoldedQuestionNode](true, defaultDuration, increment(&nodeIndex))
			node.Difficulty = difficulty
			nodeConfigs[i] = append(nodeConfigs[i], *config(node, true))
		}
	}
	return []*WorksheetSectionConfigs{
		{
			IntroductionConfigs: SectionConfig{
				LessonPlanConfigs: introductionConfigs,
			},
			NodeConfigsForSkillSections: nodeConfigs,
		},
	}, nil
}

type GetFallbackAIQuestionInput struct {
	Type       question.LessonPlanQuestionType
	Difficulty Difficulty
	Index      int
}

type FallbackAIQuestionNodes struct {
	Configs      *WorksheetSectionConfigs
	Indices      []int
	Difficulties []Difficulty
	Types        []question.LessonPlanQuestionType
}

func GetFallbackAIQuestionNodesConfig(questions []GetFallbackAIQuestionInput, preferMultiChoice bool) FallbackAIQuestionNodes {
	nodes := []LessonPlanNodeConfig{}
	types := []question.LessonPlanQuestionType{}

	createAndAppendNode := func(nodeType interface{}, difficulty Difficulty) {
		switch nodeType.(type) {
		case *MultipleChoiceQuestionNode:
			node := lpn[MultipleChoiceQuestionNode](true, sec(120), 0)
			node.Difficulty = difficulty
			nodes = append(nodes, *config(node, true))
			types = append(types, question.LessonPlanQuestionTypeMultipleChoice)
		case *ShortAnswerQuestionNode:
			node := lpn[ShortAnswerQuestionNode](true, sec(120), 0)
			node.Difficulty = difficulty
			nodes = append(nodes, *config(node, true))
			types = append(types, question.LessonPlanQuestionTypeShortAnswer)
		case *ContemplativeQuestionNode:
			node := lpn[ContemplativeQuestionNode](true, sec(120), 0)
			node.Difficulty = difficulty
			nodes = append(nodes, *config(node, true))
			types = append(types, question.LessonPlanQuestionTypeOpenEnded)
		}
	}

	for _, q := range questions {
		switch q.Type {
		case question.LessonPlanQuestionTypeShortAnswer:
			if preferMultiChoice {
				createAndAppendNode((*MultipleChoiceQuestionNode)(nil), q.Difficulty)
			} else {
				createAndAppendNode((*ShortAnswerQuestionNode)(nil), q.Difficulty)
			}
		case question.LessonPlanQuestionTypeMultipleChoice:
			createAndAppendNode((*MultipleChoiceQuestionNode)(nil), q.Difficulty)
		case question.LessonPlanQuestionTypeOpenEnded:
			createAndAppendNode((*ContemplativeQuestionNode)(nil), q.Difficulty)
		}
	}

	return FallbackAIQuestionNodes{
		Configs: &WorksheetSectionConfigs{
			NodeConfigsForSkillSections: [][]LessonPlanNodeConfig{nodes},
		},
		Indices:      lo.Map(questions, func(q GetFallbackAIQuestionInput, _ int) int { return q.Index }),
		Difficulties: lo.Map(questions, func(q GetFallbackAIQuestionInput, _ int) Difficulty { return q.Difficulty }),
		Types:        types,
	}
}

type ConfigCreatorOptions struct {
	IncludeEnabling  bool
	IncludeExtension bool
	CustomDifficulty *Difficulty
	MinDifficulty    *int
	MaxDifficulty    *int
}

func CreateWorksheetConfigCreator(totalCount int, options ConfigCreatorOptions) func(WorksheetPlanSectionType) WorksheetPlanSectionConfigsInput {
	calculateCount := func(sectionType WorksheetPlanSectionType) int {
		switch {
		case options.IncludeEnabling && options.IncludeExtension:
			half := totalCount / 2
			if sectionType == WorksheetPlanSectionTypeEnabling {
				return half + totalCount%2
			}
			return half
		case options.IncludeEnabling && sectionType == WorksheetPlanSectionTypeEnabling:
			return totalCount
		case options.IncludeExtension && sectionType == WorksheetPlanSectionTypeExtension:
			return totalCount
		default:
			return 0
		}
	}

	getDifficulty := func(sectionType WorksheetPlanSectionType) Difficulty {
		if options.CustomDifficulty != nil {
			return *options.CustomDifficulty
		}

		minDifficulty, maxDifficulty := DifficultyEnabling, DifficultyExtension
		// Convert worksheet difficulties if provided
		convertDifficulty := func(worksheetDiff *int) (Difficulty, error) {
			if worksheetDiff == nil {
				return "", fmt.Errorf("nil difficulty")
			}
			return WorksheetEditDifficulty(*worksheetDiff).ToLessonPlanDifficulty()
		}
		if diff, err := convertDifficulty(options.MinDifficulty); err == nil {
			minDifficulty = diff
		}
		if diff, err := convertDifficulty(options.MaxDifficulty); err == nil {
			maxDifficulty = diff
		}

		if sectionType == WorksheetPlanSectionTypeExtension {
			return maxDifficulty
		}
		return minDifficulty
	}

	return func(sectionType WorksheetPlanSectionType) WorksheetPlanSectionConfigsInput {
		return WorksheetPlanSectionConfigsInput{
			Type:       sectionType,
			NodeType:   LessonPlanNodeTypeScaffoldedQuestion,
			Difficulty: getDifficulty(sectionType),
			Count:      calculateCount(sectionType),
		}
	}
}

func increment(i *int) int {
	*i = *i + 1
	return *i
}

func NewLessonPlanNode(t LessonPlanNodeType) LessonPlanNode {
	var n LessonPlanNode
	switch t {
	case LessonPlanNodeTypeIntroduction:
		n = &IntroductionNode{}
	case LessonPlanNodeTypeApplication:
		n = &ApplicationNode{}
	case LessonPlanNodeTypeActivity:
		n = &ActivityNode{}
	case LessonPlanNodeTypeBrainTeaser:
		n = &BrainTeaserNode{}
	case LessonPlanNodeTypeMathsDebate:
		n = &MathsDebateNode{}
	case LessonPlanNodeTypeWarmUpQuestions:
		n = &WarmUpQuestionsNode{}
	case LessonPlanNodeTypeWarmUpQuestionsWithContext:
		n = &WarmUpQuestionsWithContextNode{}
	case LessonPlanNodeTypeThoughtSparker:
		n = &ThoughtSparkerNode{}
	case LessonPlanNodeTypeExitTicket:
		n = &ExitTicketNode{}
	case LessonPlanNodeTypeContemplativeQuestion:
		n = &ContemplativeQuestionNode{
			Difficulty: DifficultyMild,
		}
	case LessonPlanNodeTypeLessonAgenda:
		n = &LessonAgendaNode{}
	case LessonPlanNodeTypeLearningGoals:
		n = &LearningGoalsNode{}
	case LessonPlanNodeTypeSituationalQuestion:
		n = &SituationalQuestionNode{}
	case LessonPlanNodeTypeScaffoldedQuestion:
		n = &ScaffoldedQuestionNode{
			Difficulty: DifficultyMild,
		}
	case LessonPlanNodeTypeMultipleChoiceQuestion:
		n = &MultipleChoiceQuestionNode{}
	case LessonPlanNodeTypeShortAnswerQuestion:
		n = &ShortAnswerQuestionNode{}
	case LessonPlanNodeTypeMisconceptionQuestion:
		n = &MisconceptionQuestionNode{}
	}
	base := n.Base()
	base.SetID(id.MustNew())
	base.Type = t
	return n
}

func (t LessonPlanNodeType) SectionLabel() string {
	switch t {
	case LessonPlanNodeTypeApplication, LessonPlanNodeTypeWarmUpQuestions, LessonPlanNodeTypeLearningGoals, LessonPlanNodeTypeLessonAgenda, LessonPlanNodeTypeWarmUpQuestionsWithContext, LessonPlanNodeTypeIntroduction:
		return "IntroductionSection"
	case LessonPlanNodeTypeContemplativeQuestion,
		LessonPlanNodeTypeThoughtSparker,
		LessonPlanNodeTypeExitTicket,
		LessonPlanNodeTypeActivity,
		LessonPlanNodeTypeMultipleChoiceQuestion,
		LessonPlanNodeTypeShortAnswerQuestion,
		LessonPlanNodeTypeSituationalQuestion,
		LessonPlanNodeTypeScaffoldedQuestion,
		LessonPlanNodeTypeMathsDebate,
		LessonPlanNodeTypeMisconceptionQuestion:
		return "PracticeSection"
	case LessonPlanNodeTypeSkills, LessonPlanNodeTypeSkillSlide,
		LessonPlanNodeTypeStudentQuestions, LessonPlanNodeTypeTeacherQuestions, LessonPlanNodeTypeBrainTeaser:
		return "SkillSection"
	default:
		return ""
	}
}

func (i UpdateLessonPlanCardNodeInput) Validate(create bool) error {
	if create {
		if i.Title == nil || *i.Title == "" {
			return fmt.Errorf("title is required")
		}
		if len(i.Icon) == 0 && i.Type != LessonPlanCardGeogebraSimulationType {
			return fmt.Errorf("icon is required")
		}
	}
	switch i.Type {
	case LessonPlanCardBlankNoteType:
		return nil
	case LessonPlanCardEmbedType:
		if i.EmbedInput == nil {
			return fmt.Errorf("embedInput is required")
		}
	case LessonPlanCardFileType:
		if i.FileInput == nil {
			return fmt.Errorf("fileInput is required")
		}
	case LessonPlanCardVideoType:
		if i.VideoInput == nil {
			return fmt.Errorf("videoInput is required")
		}
	case LessonPlanCardGeogebraSimulationType:
		if i.GeogebraInput == nil {
			return fmt.Errorf("geogebraInput is required")
		}
	}
	return nil
}

// Creates a concrete implementation of LessonPlanCard based on the input
// UpdateLessonPlanCardNodeInput.Validate() must be called before this
func (i UpdateLessonPlanCardNodeInput) CreateCard(nodeID *id.ID) LessonPlanCard {
	baseCard := BaseCard{
		BaseNode: BaseNode{
			Ready:   true,
			Enabled: true,
		},
	}
	if nodeID == nil || *nodeID == "" {
		baseCard.SetID(id.MustNew())
	} else {
		baseCard.SetID(*nodeID)
	}
	if i.Title != nil {
		baseCard.Title = *i.Title
	}
	if i.Image != nil {
		baseCard.Image = *i.Image
	}
	if len(i.Icon) > 0 {
		baseCard.Icon = i.Icon
	}

	var card LessonPlanCard
	switch i.Type {
	case LessonPlanCardBlankNoteType:
		card = &BlankNoteCardNode{}
	case LessonPlanCardEmbedType:
		card = &EmbedCardNode{
			URL: i.EmbedInput.URL,
		}
	case LessonPlanCardFileType:
		card = &FileCardNode{
			URL: i.FileInput.URL,
		}
	case LessonPlanCardVideoType:
		card = &VideoCardNode{
			URL: i.VideoInput.URL,
		}
	case LessonPlanCardGeogebraSimulationType:
		card = &GeogebraSimulationCardNode{
			SimulationType: i.GeogebraInput.SimulationType,
			Content:        i.GeogebraInput.Content,
		}
	}

	card.SetBaseCard(baseCard)
	return card
}

	LessonPlanEditLevelLevelThree LessonPlanEditLevel = "LEVEL_THREE"
	LessonPlanEditLevelLevelFour  LessonPlanEditLevel = "LEVEL_FOUR"
	LessonPlanEditLevelLevelFive  LessonPlanEditLevel = "LEVEL_FIVE"
)

var LessonPlanEditLevels = []LessonPlanEditLevel{
	LessonPlanEditLevelLevelOne,
	LessonPlanEditLevelLevelTwo,
	LessonPlanEditLevelLevelThree,
	LessonPlanEditLevelLevelFour,
	LessonPlanEditLevelLevelFive,
}

// ToRelativeAdjustment converts LessonPlanEditLevel enum values to relative integer adjustments
// which is used to adjust the length and emoji count of the content
func (level LessonPlanEditLevel) ToRelativeAdjustment() int {
	switch level {
	case LessonPlanEditLevelLevelOne:
		return -2
	case LessonPlanEditLevelLevelTwo:
		return -1
	case LessonPlanEditLevelLevelThree:
		return 0
	case LessonPlanEditLevelLevelFour:
		return 1
	case LessonPlanEditLevelLevelFive:
		return 2
	default:
		return 0 // Default to neutral
	}
}

// ToTranslation converts LessonPlanEditLevel enum values to relative string adjustments
// which is used to adjust the difficulty and reading level of the content
func (level LessonPlanEditLevel) ToTranslation() string {
	switch level {
	case LessonPlanEditLevelLevelOne:
		return "2 levels lower"
	case LessonPlanEditLevelLevelTwo:
		return "1 level lower"
	case LessonPlanEditLevelLevelThree:
		return "same level"
	case LessonPlanEditLevelLevelFour:
		return "1 level higher"
	case LessonPlanEditLevelLevelFive:
		return "2 levels higher"
	default:
		return "same level"
	}

type AddNodeInput struct {
	SectionID            id.ID                     `json:"sectionID"`
	Type                 *LessonPlanNodeType       `json:"type"`
	Title                *string                   `json:"title,omitempty"`
	SelectedContext      *string                   `json:"selectedContext,omitempty"`
	LessonPlanCardInputs []LessonPlanCardNodeInput `json:"lessonPlanCardInputs"`
	StudentQuestions     []string                  `json:"studentQuestions"`
	TeacherQuestions     []string                  `json:"teacherQuestions"`
	BrainTeasers         []string                  `json:"brainTeasers"`
	SkillSlides          []string                  `json:"skillSlides"`
	Image                *string                   `json:"image,omitempty"`
}

func (i AddNodeInput) Validate() error {
	if i.IsEmpty() {
		return fmt.Errorf("at least one field must be provided")
	}

	// * Make sure either type is provided or other inputs are provided
	if i.Type != nil && (len(i.LessonPlanCardInputs) > 0 || len(i.StudentQuestions) > 0 || len(i.TeacherQuestions) > 0 || len(i.BrainTeasers) > 0 || len(i.SkillSlides) > 0) {
		return fmt.Errorf("either type or other inputs must be provided")
	}

	return nil
}

func (i AddNodeInput) IsEmpty() bool {
	return i.Type == nil && len(i.LessonPlanCardInputs) == 0 && len(i.StudentQuestions) == 0 && len(i.TeacherQuestions) == 0 && len(i.BrainTeasers) == 0 && len(i.SkillSlides) == 0
}

func (i AddNodeInput) RequiresContext() bool {
	//* All types require context except for LearningGoals, LessonAgenda
	return i.Type != nil && !lo.Contains(LessonGoalsTypes, *i.Type)
}

type WorksheetPlanConfigurationInput struct {
	Type                      resource.WorksheetPlanType    `json:"type"`
	QuestionCount             int                           `json:"questionCount"`
	AverageQuestionDifficulty float64                       `json:"averageQuestionDifficulty"`
	WorkingOutSpaceEnabled    bool                          `json:"workingOutSpaceEnabled"`
	ReducePaperEnabled        bool                          `json:"reducePaperEnabled"`
	WorksheetLength           int                           `json:"worksheetLength"`
	TextSize                  int                           `json:"textSize"`
	ImageSize                 int                           `json:"imageSize"`
	SectionHeadersEnabled     bool                          `json:"sectionHeadersEnabled"`
	QuestionDifficultyEnabled bool                          `json:"questionDifficultyEnabled"`
	ShowAnswerArea            bool                          `json:"showAnswerArea"`
	ShowClassName             bool                          `json:"showClassName"`
	ExplanationSpace          int                           `json:"explanationSpace"`
	QuestionTypeEnabled       bool                          `json:"questionTypeEnabled"`
	AnswersEnabled            bool                          `json:"answersEnabled"`
	NodeCounts                []WorksheetPlanNodeCountInput `json:"nodeCounts"`
	MaxDifficulty             *int                          `json:"maxDifficulty"`
	MinDifficulty             *int                          `json:"minDifficulty"`
}

func (c *WorksheetPlanConfigurationInput) Validate() error {
	if c.MinDifficulty == nil || c.MaxDifficulty == nil {
		return oops.Errorf("worksheets must have min and max difficulty")
	}
	if *c.MinDifficulty > *c.MaxDifficulty {
		return oops.Errorf("min difficulty must be less than max difficulty")
	}
	if len(c.NodeCounts) == 0 {
		return oops.Errorf("at least one node count must be provided")
	}
	return nil
}

func (w WorksheetPlanConfigurationInput) LessonPlanNodeCountMap() map[LessonPlanNodeType]int {
	m := make(map[LessonPlanNodeType]int)
	for _, n := range w.NodeCounts {
		if n.NodeType != nil {
			m[*n.NodeType] += n.Count
		}
	}
	return m
}

func (w WorksheetPlanConfigurationInput) PopulateQuestionCount(config *WorksheetPlanConfiguration) *WorksheetPlanConfiguration {
	for _, n := range w.NodeCounts {
		if n.QuestionType == nil {
			continue
		}
		switch *n.QuestionType {
		case WorksheetQuestionTypeMultipleChoice:
			config.MultiQuestionCount += n.Count
		case WorksheetQuestionTypeShortAnswer:
			config.ShortAnswerQuestionCount += n.Count
		case WorksheetQuestionTypeOpenEnded:
			config.OpenEndedQuestionCount += n.Count
		}
	}
	return config
}

// api/lesson_plan/worksheet_edit.go
package lessonplan

import (
	"github.com/MathGaps/resources/api/question"
	"github.com/samber/lo"
	"github.com/samber/oops"
)

// WorksheetEditDifficulty represents ordered difficulty levels.
// 0: ENABLING, 1: EASY, 2: MEDIUM, 3: HARD, 4: EXTENSION
type WorksheetEditDifficulty int

const (
	WorksheetEditDifficultyEnabling WorksheetEditDifficulty = 0
	WorksheetEditDifficultyEasy     WorksheetEditDifficulty = 1
	WorksheetEditDifficultyMedium   WorksheetEditDifficulty = 2
	WorksheetEditDifficultyHard     WorksheetEditDifficulty = 3
	WorksheetEditDifficultyExt      WorksheetEditDifficulty = 4
)

func (d WorksheetEditDifficulty) ToLessonPlanDifficulty() (Difficulty, error) {
	switch d {
	case WorksheetEditDifficultyEnabling:
		return DifficultyEnabling, nil
	case WorksheetEditDifficultyEasy:
		return DifficultyMild, nil
	case WorksheetEditDifficultyMedium:
		return DifficultyHot, nil
	case WorksheetEditDifficultyHard:
		return DifficultySpicy, nil
	case WorksheetEditDifficultyExt:
		return DifficultyExtension, nil
	default:
		return "", oops.Errorf("invalid lesson plan difficulty for worksheet difficulty `%d`", d)
	}
}

func (d WorksheetEditDifficulty) ToQuestionDifficulty() (question.QuestionDifficulty, error) {
	switch d {
	case WorksheetEditDifficultyEasy:
		return question.QuestionDifficultyEasy, nil
	case WorksheetEditDifficultyMedium:
		return question.QuestionDifficultyMedium, nil
	case WorksheetEditDifficultyHard:
		return question.QuestionDifficultyHard, nil
	case WorksheetEditDifficultyExt:
		return question.QuestionDifficultyExtension, nil
	default:
		return "", oops.Errorf("unknown question difficulty for worksheet difficulty %d", d)
	}
}

func WorksheetEditDifficultyFromQuestionDifficulty(d question.QuestionDifficulty) (WorksheetEditDifficulty, error) {
	switch d {
	case question.QuestionDifficultyEasy:
		return WorksheetEditDifficultyEasy, nil
	case question.QuestionDifficultyMedium:
		return WorksheetEditDifficultyMedium, nil
	case question.QuestionDifficultyHard:
		return WorksheetEditDifficultyHard, nil
	case question.QuestionDifficultyExtension:
		return WorksheetEditDifficultyExt, nil
	default:
		return 0, oops.Errorf("unknown question difficulty %s", d)
	}
}

func WorksheetEditDifficultyFromLessonPlanDifficulty(d Difficulty) (WorksheetEditDifficulty, error) {
	switch d {
	case DifficultyEnabling:
		return WorksheetEditDifficultyEnabling, nil
	case DifficultyMild:
		return WorksheetEditDifficultyEasy, nil
	case DifficultyHot:
		return WorksheetEditDifficultyMedium, nil
	case DifficultySpicy:
		return WorksheetEditDifficultyHard, nil
	case DifficultyExtension:
		return WorksheetEditDifficultyExt, nil
	default:
		return 0, oops.Errorf("unknown lesson plan difficulty %s", d)
	}
}

func WorksheetEditDifficultyFromString(s string) (WorksheetEditDifficulty, error) {
	lpDifficulty := Difficulty(s)
	if lo.Contains(WorksheetPlanDifficulties, lpDifficulty) {
		return WorksheetEditDifficultyFromLessonPlanDifficulty(lpDifficulty)
	}

	qDifficulty := question.QuestionDifficulty(s)
	if lo.Contains(question.QuestionDifficulties, qDifficulty) {
		return WorksheetEditDifficultyFromQuestionDifficulty(qDifficulty)
	}

	return 0, oops.Errorf("invalid worksheet edit difficulty: `%s`", s)
}

func StringAsLessonPlanDifficulty(s string) (Difficulty, error) {
	if wsDifficulty, err := WorksheetEditDifficultyFromString(s); err != nil {
		return "", err
	} else {
		return wsDifficulty.ToLessonPlanDifficulty()
	}
}

func CompareLessonPlanDifficulties(a, b Difficulty) (int, error) {
	wsDiffA, err := WorksheetEditDifficultyFromLessonPlanDifficulty(a)
	if err != nil {
		return 0, err
	}
	wsDiffB, err := WorksheetEditDifficultyFromLessonPlanDifficulty(b)
	if err != nil {
		return 0, err
	}
	return wsDiffA.Compare(wsDiffB)
}

func (d WorksheetEditDifficulty) Compare(difficulty WorksheetEditDifficulty) (int, error) {
	a, b := int(d), int(difficulty)
	switch {
	case a < b:
		return -1, nil
	case a > b:
		return 1, nil
	default:
		return 0, nil
	}
}
