func (i UpsertFolderPreviewInput) QdrantPayload() (map[string]*qdrant.Value, error) {
	if i.UUID == "" {
		return nil, errors.New("UUID is required to upsert folder preview")
	}

	payload := map[string]*qdrant.Value{}

	payload["resourceType"] = qdrant.NewValueString("FOLDER")
	if i.ID != "" {
		payload["id"] = qdrant.NewValueString(i.ID)
	}
	if i.Name != nil && *i.Name != "" {
		payload["name"] = qdrant.NewValueString(*i.Name)
	}
	if i.Path != nil && *i.Path != "" {
		payload["path"] = qdrant.NewValueString(*i.Path)
	}
	if i.ClassID != nil && *i.ClassID != "" {
		payload["classID"] = qdrant.NewValueString(*i.ClassID)
	}
	if i.CreatedAt != nil {
		payload["createdAt"] = qdrant.NewValueString(i.CreatedAt.Format(time.RFC3339))
	}
	if i.LastViewedAt != nil {
		payload["lastViewedAt"] = qdrant.NewValueString(i.LastViewedAt.Format(time.RFC3339))
	}

	if i.Topics != nil && len(*i.Topics) > 0 {
		topicIDs := lo.Map(*i.Topics, func(id id.ID, _ int) *qdrant.Value {
			return qdrant.NewValueString(id)
		})
		payload["topicIDs"] = qdrant.NewValueList(&qdrant.ListValue{Values: topicIDs})
	}

	if i.Subtopics != nil && len(*i.Subtopics) > 0 {
		subtopicIDs := lo.Map(*i.Subtopics, func(id id.ID, _ int) *qdrant.Value {
			return qdrant.NewValueString(id)
		})
		payload["subtopicIDs"] = qdrant.NewValueList(&qdrant.ListValue{Values: subtopicIDs})
	}

	if i.ResourceTypes != nil && len(*i.ResourceTypes) > 0 {
		resourceTypes := lo.Map(*i.ResourceTypes, func(rt resource.ResourceType, _ int) *qdrant.Value {
			return qdrant.NewValueString(string(rt))
		})
		payload["resourceTypes"] = qdrant.NewValueList(&qdrant.ListValue{Values: resourceTypes})
	}

	return payload, nil
}

type SearchFolderFilter struct {
	ClassIDs       []id.ID
	Search         string  `json:"search"`
	Path           string  `json:"path"`
	Recursive      bool    `json:"recursive"`
	ScoreThreshold float64 `json:"scoreThreshold,omitempty"`
}

func (f *SearchFolderFilter) ToQdrantFilter() *qdrant.Filter {
	must := []*qdrant.Condition{}

	if len(f.ClassIDs) > 0 {
		must = append(must, &qdrant.Condition{
			ConditionOneOf: &qdrant.Condition_Field{
				Field: &qdrant.FieldCondition{
					Key: "classID",
					Match: &qdrant.Match{
						MatchValue: &qdrant.Match_Keywords{
							Keywords: &qdrant.RepeatedStrings{
								Strings: f.ClassIDs,
							},
						},
					},
				},
			},
		})
	}

	if f.Path != "" {
		path := common.EnsurePathFormat(f.Path)
		if f.Recursive {
			must = append(must, &qdrant.Condition{
				ConditionOneOf: &qdrant.Condition_Field{
					Field: &qdrant.FieldCondition{
						Key: "path",
						Match: &qdrant.Match{
							MatchValue: &qdrant.Match_Text{
								Text: path,
							},
						},
					},
				},
			})
		} else {
			must = append(must, qdrant.NewMatch("path", path))
		}
	}

	return &qdrant.Filter{Must: must}
}
