// Ensure all events implement natstooling.Event interface
var (
	_ natstooling.Event = (*DiagnosticTestSessionStatusUpdated)(nil)
	_ natstooling.Event = (*DiagnosticTestSessionUpdated)(nil)
	_ natstooling.Event = (*LessonPlanDocumentationUpdated)(nil)
	_ natstooling.Event = (*LessonPlanNodeContextUpdated)(nil)
	_ natstooling.Event = (*DiagnosticTestSessionProgressChanged)(nil)
	_ natstooling.Event = (*DiagnosticTestSessionQuestionAnswered)(nil)
	_ natstooling.Event = (*LessonPlanCoverImageRequest)(nil)
	_ natstooling.Event = (*LessonPlanCoverImageGenerated)(nil)
	_ natstooling.Event = (*ClassStudentsReadyForTest)(nil)
	_ natstooling.Event = (*BulkResourceDuplicationStarted)(nil)
	_ natstooling.Event = (*BulkResourceDuplicationCompleted)(nil)
	_ natstooling.Event = (*BulkResourceDeletionStarted)(nil)
	_ natstooling.Event = (*BulkResourceDeletionCompleted)(nil)
	_ natstooling.Event = (*LessonReadyUpdated)(nil)
	_ natstooling.Event = (*LessonPlanReadyUpdated)(nil)
	_ natstooling.Event = (*LessonPlanUpdated)(nil)
)

// DiagnosticTestSessionUpdated represents the session after updating its properties
type DiagnosticTestSessionUpdated struct {
	SessionID id.ID `json:"id"`
}

// DiagnosticTestSessionStatusUpdated represents the diagnostic test session when the status changed
type DiagnosticTestSessionStatusUpdated struct {
	SessionID id.ID `json:"id"`
}

type DiagnosticTestSessionProgressChanged struct {
	SessionID id.ID `json:"id"`
}

type DiagnosticTestSessionQuestionAnswered struct {
	SessionID string `json:"id"`
}

type ActiveDiagnosticTestForStudentUpdated struct {
	StudentID id.ID `json:"id"`
}

type ClassStudentsReadyForTest struct {
	ClassID id.ID `json:"id"`
}

type StudentProgressesForSession struct {
	SessionID  id.ID                                    `json:"sessionID"`
	Progresses []*diagnostictestsession.StudentProgress `json:"progresses"`
}

func (s StudentProgressesForSession) IsEntity() {}

type AnsweredQuestionsForSession struct {
	SessionID         id.ID                                             `json:"sessionID"`
	AnsweredQuestions []*diagnostictest.DiagnosticTestAnsweredQuestions `json:"answeredQuestions"`
}

func (a AnsweredQuestionsForSession) IsEntity() {}

// BulkResourceDuplicationStarted represents the start of a bulk resource duplication operation
type BulkResourceDuplicationStarted struct {
	RequestID     string                          `json:"requestID"`
	UserID        id.ID                           `json:"userID"`
	ResourceIDs   []id.ID                         `json:"resourceIDs"`
	ClassIDs      []id.ID                         `json:"classIDs"`
	FolderDetails *resource.SelectedFolderDetails `json:"folderDetails,omitempty"`
}

func (a BulkResourceDuplicationStarted) IsEntity() {}

// BulkResourceDuplicationCompleted represents the completion of a bulk resource duplication operation
type BulkResourceDuplicationCompleted struct {
	RequestID     string  `json:"requestID"`
	OriginalIDs   []id.ID `json:"originalIDs"`
	DuplicatedIDs []id.ID `json:"duplicatedIDs"`
	FailedIDs     []id.ID `json:"failedIDs"`
}

func (b BulkResourceDuplicationCompleted) IsEntity() {}

// BulkResourceDeletionStarted represents the start of a bulk resource deletion operation
type BulkResourceDeletionStarted struct {
	RequestID   string  `json:"requestID"`
	UserID      id.ID   `json:"userID"`
	ResourceIDs []id.ID `json:"resourceIDs"`
}

func (a BulkResourceDeletionStarted) IsEntity() {}

// BulkResourceDeletionCompleted represents the completion of a bulk resource deletion operation
type BulkResourceDeletionCompleted struct {
	RequestID  string  `json:"requestID"`
	DeletedIDs []id.ID `json:"deletedIDs"`
	FailedIDs  []id.ID `json:"failedIDs"`
}

func (b BulkResourceDeletionCompleted) IsEntity() {}

// LessonPlanDocumentationUpdated represents the lesson plan documentation node updated
type LessonPlanDocumentationUpdated struct {
	LessonPlanID id.ID  `json:"lessonPlanID"`
	LessonID     id.ID  `json:"lessonID"`
	NodeType     string `json:"nodeType"`
}

// LessonPlanNodeContextUpdated represents the context update for a lesson plan node
type LessonPlanNodeContextUpdated struct {
	LessonPlanID id.ID  `json:"lessonPlanID"`
	NodeID       id.ID  `json:"nodeID"`
	Context      string `json:"context"`
}

// TODO: Remove below types once teaching and diagnostics migrated to fx
type TestQuestionAnswered struct {
	TestID     id.ID  `json:"testID"`
	StudentID  id.ID  `json:"studentID"`
	Result     bool   `json:"result"`
	QuestionID id.ID  `json:"questionID"`
	Answer     string `json:"answer"`
	TimeTaken  int64  `json:"timeTaken"`
}

type TestFinished struct {
	TestID id.ID `json:"testID"`
}

type LessonPlanCoverImageRequest struct {
	UserID       id.ID `json:"userID"`
	LessonPlanID id.ID `json:"lessonPlanID"`
	ClassID      id.ID `json:"classID"`
	NodeID       id.ID `json:"nodeID"`
}

type LessonPlanCoverImageGenerated struct {
	UserID       id.ID  `json:"userID"`
	LessonPlanID id.ID  `json:"lessonPlanID"`
	ImageUrl     string `json:"imageUrl"`
}

type LessonReadyUpdated struct {
	LessonID     id.ID `json:"lessonID"`
	LessonPlanID id.ID `json:"lessonPlanID"`
}

type LessonPlanReadyUpdated struct {
	LessonPlanID id.ID `json:"lessonPlanID"`
}

type LessonPlanUpdated struct {
	LessonPlanID id.ID                              `json:"lessonPlanID"`
	UserID       id.ID                              `json:"userID"`
	Input        lessonplan.UpdateLessonPlanInputV3 `json:"input"`
	IsClass      bool                               `json:"isClass"`
}

// api/events/mixpanel.go
package events

import (
	"time"

	"github.com/MathGaps/core/pkg/mixpanel"
)

type ResourceType string

const (
	LessonPlanType     ResourceType = "Lesson Plan"
	WorksheetType      ResourceType = "Worksheet"
	DiagnosticTestType ResourceType = "Assessment"
)

type ResourceEvent struct {
	Year         string       `json:"year"`
	SkillCount   int          `json:"skillCount"`
	ClassName    string       `json:"class"`
	ResourceType ResourceType `json:"type"`
}

type LessonPlanCreatedEvent struct {
	mixpanel.MixpanelEvent `mixpanel:"Resource Created"`
	ResourceEvent
	LessonCount int `json:"lessonCount"`
}

type WorksheetCreatedEvent struct {
	mixpanel.MixpanelEvent `mixpanel:"Resource Created"`
	ResourceEvent
}

type DiagnosticTestCreatedEvent struct {
	mixpanel.MixpanelEvent `mixpanel:"Resource Created"`
	ResourceEvent
}

type DiagnosticTestFinishedEvent struct {
	mixpanel.MixpanelEvent `mixpanel:"Assessment Finished"`
	Duration               time.Duration `json:"duration"`
	Year                   string        `json:"year"`
	SkillCount             int           `json:"skillCount"`
	ClassName              string        `json:"class"`
}
