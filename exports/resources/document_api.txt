// api/document/repository.go
package document

import (
	"context"

	"github.com/MathGaps/core/pkg/domain/id"
	"github.com/MathGaps/resources/api/common"
	"github.com/MathGaps/resources/api/skill"
)

type Repository interface {
	Get(ctx context.Context, documentID id.ID) (*Document, error)
	CreateForIndividual(ctx context.Context, input CreateDocumentInput) (*Document, error)
	CreateForClass(ctx context.Context, classID id.ID, input CreateDocumentInput) (*Document, error)
	Update(ctx context.Context, documentID id.ID, input UpdateDocumentInput) (*Document, error)
	DeleteDocument(ctx context.Context, documentID id.ID) (bool, error)
	GenerateUnitPlanContent(ctx context.Context, input GenerateUnitPlanInput) (<-chan *DocumentContentChunk, error)
	GenerateDocumentContent(ctx context.Context, input GenerateDocumentContentInput) (<-chan *DocumentContentChunk, error)
	EditDocument(ctx context.Context, input EditDocumentInput) (<-chan *DocumentContentChunk, error)
	ChangeSkills(ctx context.Context, input ChangeSkillsInput) (<-chan *DocumentContentChunk, error)
	AdjustDocumentCount(ctx context.Context, input AdjustCountInput) (<-chan *DocumentContentChunk, error)
	SliderAdjustment(ctx context.Context, input SliderAdjustmentInput) (<-chan *DocumentContentChunk, error)
	EditSpecificText(ctx context.Context, input EditSpecificTextInput) (*EditSpecificTextOutput, error)
	GetDirectSkills(ctx context.Context, documentID id.ID) ([]*skill.Skill, error)
	SearchSkills(ctx context.Context, documentID id.ID, query string, options common.OffsetPaginationOptions, scoreThreshold float64) ([]*skill.Skill, error)
	UpdateCoverImage(ctx context.Context, documentID id.ID, imageUrl string) (*Document, error)
	TransferDocumentToClass(ctx context.Context, teacherID, classID id.ID) error
	DuplicateDocumentToClass(ctx context.Context, documentID id.ID, classID id.ID, folderID id.ID) (*Document, error)
}
