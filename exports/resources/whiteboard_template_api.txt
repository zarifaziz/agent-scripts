// api/whiteboard_template/entity.go
package whiteboardtemplate

import (
	"github.com/MathGaps/core/pkg/domain/id"
)

type WhiteboardTemplateDoc struct {
	ID       id.ID          `firestore:"id"`
	Template map[string]any `firestore:"template"`
	Context  string         `firestore:"context"`
}

type SkillWhiteboardTemplate struct {
	ID         id.ID   `firestore:"id"`
	SkillID    string  `firestore:"skillId"`
	TemplateID *string `firestore:"templateId"`
	CreatedAt  string  `firestore:"createdAt"`
}

type GeneralWhiteboardTemplate struct {
	ID      id.ID          `firestore:"id"`
	Name    string         `firestore:"name"`
	IconMap map[string]any `firestore:"iconMap"`
}

type SkillSlide struct {
	SkillTemplateID    id.ID          `json:"skillTemplateID"`
	GeneralTemplateID  id.ID          `json:"generalTemplateID"`
	Title              string         `json:"title"`
	Icon               map[string]any `json:"icon"`
	WhiteboardTemplate map[string]any `json:"whiteboardTemplate"`
}

const (
	OverviewGeneralTemplateID     = "overview_general_template"
	OverviewSkillTemplateIDPrefix = "overview_skill_template_"
)

func CreateOverviewSkillTemplateID(skillID string) string {
	return OverviewSkillTemplateIDPrefix + skillID
}

// api/whiteboard_template/repository.go
package whiteboardtemplate

import (
	"context"

	"github.com/MathGaps/core/pkg/domain/id"
)

type Repository interface {
	GetWhiteboardTemplate(ctx context.Context, id id.ID) (template map[string]any, ready bool, err error)
	GetSkillsTemplatesMap(ctx context.Context, skillIDs []id.ID) (map[id.ID][]*SkillWhiteboardTemplate, error)
	GetGeneralTemplate(ctx context.Context, id id.ID) (*GeneralWhiteboardTemplate, error)
	GetSkillSlides(ctx context.Context, skillID id.ID, excludedSkillTemplateIDs []id.ID) ([]*SkillSlide, error)
	GetSkillSlidesForLessonPlan(ctx context.Context, lessonPlanID, skillID id.ID) ([]*SkillSlide, error)
	ResolveSkillSlides(ctx context.Context, skillTemplateIDs []id.ID, generalTemplateIDs []id.ID) ([]*SkillSlide, error)
}
