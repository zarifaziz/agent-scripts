# Command patterns that always prompt for confirmation
# Use regex patterns

# brew package management
^brew (install|uninstall|remove|upgrade)

# git destructive operations
^git checkout .
^git checkout -f
^git checkout --force
^git reset
^git clean
^git stash drop
